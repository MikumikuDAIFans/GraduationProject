```python
# 冲突检测核心函数
class SchedulingEngine:
    def detect_conflicts(self, existing_events: List[Event], new_event: Event) -> List[Event]:
        """检测日程冲突"""
        conflicts = []
        
        # 计算新事件的有效时间范围（包括缓冲时间）
        new_start = new_event.start_time - timedelta(minutes=new_event.buffer_before)
        new_end = new_event.end_time + timedelta(minutes=new_event.buffer_after)
        new_slot = TimeSlot(new_start, new_end)
        
        for event in existing_events:
            # 跳过已取消的事件
            if event.id == new_event.id:
                continue
                
            # 计算现有事件的有效时间范围
            event_start = event.start_time - timedelta(minutes=event.buffer_before)
            event_end = event.end_time + timedelta(minutes=event.buffer_after)
            event_slot = TimeSlot(event_start, event_end)
            
            # 检查时间冲突
            if new_slot.overlaps(event_slot):
                conflicts.append(event)
        
        return conflicts
```

```python
        # 室内关键词（排除）
        indoor_keywords = [
            "室内", "家里", "online", "线上", "视频", "电话", "会议", "网课"
        ]
        
        message_lower = user_message.lower()
        
        # 检查室内关键词（不需要天气）
        for keyword in indoor_keywords:
            if keyword in message_lower:
                return False
        
        # 检查户外关键词
        for keyword in outdoor_keywords:
            if keyword in message_lower:
                return True
        
        return False
```

### B.1 核心数据表结构

```sql
-- 用户表
CREATE TABLE tb_users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(100),
    timezone VARCHAR(50) DEFAULT 'Asia/Shanghai',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_email (email)
);

-- 日程事件表
CREATE TABLE tb_schedule_events (
    id CHAR(36) PRIMARY KEY,
    user_id INT NOT NULL,
    title VARCHAR(200) NOT NULL,
    start_time DATETIME NOT NULL,
    end_time DATETIME NOT NULL,
    location_name VARCHAR(200),
    event_type VARCHAR(50) DEFAULT 'appointment',
    is_fixed BOOLEAN DEFAULT FALSE,
    status ENUM('planned', 'ongoing', 'completed', 'cancelled') DEFAULT 'planned',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES tb_users(id) ON DELETE CASCADE,
    INDEX idx_user_time (user_id, start_time, end_time)
);

-- 待办事项表
CREATE TABLE tb_tasks (
    id CHAR(36) PRIMARY KEY,
    user_id INT NOT NULL,
    title VARCHAR(200) NOT NULL,
    estimated_duration INT,
    priority INT DEFAULT 3,
    deadline DATETIME,
    status ENUM('pending', 'in_progress', 'completed', 'cancelled') DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES tb_users(id) ON DELETE CASCADE,
    INDEX idx_user_status (user_id, status)
);
```
